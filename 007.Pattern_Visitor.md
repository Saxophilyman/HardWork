В общем случае необходимо реализовать ряд коллекций. Пусть это будет выражение SQL, которое можно разделить по нескольким условиям

Каким бы способом я действовал в общем случае? Наверное, я постарался бы их просто расширить, создав родителя и переопределив метод (в данном случае всего один) в каждом классе.

```java
abstract class Expression {
abstract String toSql();
}
```

```java
public class EqualsExpression extends Expression {
private final String field;
private final String value;

    public EqualsExpression(String field, String value) {
        this.field = field;
        this.value = value;
    }

    @Override
    String toSql(){
        return field + " = '" + value + "'";
    }
}
```

```java
class GreaterThanExpression extends Expression {
    private final String field;
    private final int value;

    public GreaterThanExpression(String field, int value) {
        this.field = field;
        this.value = value;
    }

    @Override
    String toSql() {
        return field + " > " + value;
    }
}
```

```java
public class AndExpression extends Expression {
    private final Expression left;
    private final Expression right;

    public AndExpression(Expression left, Expression right) {
        this.left = left;
        this.right = right;
    }

    @Override
    String toSql() {
        return "(" + left.toSql() + " AND " + right.toSql() + ")";
    }
}
```

```java
public class Main {
    public static void main(String[] args) {
        Expression expression = new AndExpression(
                new GreaterThanExpression("age", 20), new EqualsExpression("name", "Cao Cao"));
        System.out.println(expression.toSql());
    }
}
```

```java
//1. интерфейс визитера
public interface ExpressionVisitor {
    String visit(EqualsExpression expression);

    String visit(GreaterThanExpression expression);

    String visit(AndExpression expression);
}
```

```java
// 2. Интерфейс самих элементов
public interface Expression {
    String accept(ExpressionVisitor visitor);
}
```

```java
//3. Добавляем метод из визитера
public class EqualsExpression implements Expression {
    private final String field;
    private final String value;

    public EqualsExpression(String field, String value) {
        this.field = field;
        this.value = value;
    }

    public String getField() {
        return field;
    }

    public String getValue() {
        return value;
    }

    @Override
    public String accept(ExpressionVisitor visitor) {
        return visitor.visit(this);
    }
}
```

```java
//5. Добавляем метод из визитера
public class AndExpression implements Expression {
    private final Expression left;
    private final Expression right;

    public AndExpression(Expression left, Expression right) {
        this.left = left;
        this.right = right;
    }

    public Expression getRight() {
        return right;
    }

    public Expression getLeft() {
        return left;
    }

    @Override
    public String accept(ExpressionVisitor visitor) {
        return visitor.visit(this);
    }
}
```

```java
//6. Для каждого нового поведения создайте конкретный класс посетителя.
public class SqlExpressionVisitor implements ExpressionVisitor{

    @Override
    public String visit(EqualsExpression expression) {
        return expression.getField() + " = '" + expression.getValue() + "'";
    }

    @Override
    public String visit(GreaterThanExpression expression) {
        return expression.getField() + " > " + expression.getValue();
    }

    @Override
    public String visit(AndExpression expression) {
        String leftSql = expression.getLeft().accept(this);
        String rightSql = expression.getRight().accept(this);

        return "(" + leftSql + " AND " + rightSql + ")";
    }
}
```

```java
public class Main {
    public static void main(String[] args) {
        Expression expression = new AndExpression(
                new GreaterThanExpression("age", 20),
                new EqualsExpression("name", "Cao Cao")
        );

        ExpressionVisitor visitor = new SqlExpressionVisitor();
        System.out.println(expression.accept(visitor));
    }
}
```

Частое слово "поведение" в данном паттерне в какой-то момент превращается в осознание концептуального разделения 
между самими данными и операциями над ними. Говоря о добавлении новых операций, с данным паттерном 
всё это выглядит очень выгодным лайфхаком.

С другой стороны под это всё можно добавить и новое выражение, аккуратно интегрировав его, но это будет более трудоёмко. 
Но если паттернов Visiter над ними больше, чем один и структура иерархии ещё может меняться... 
ещё нужно подумать как в таком случае поступать
