Java изначально предполагает смешанный стиль, где данные и работа над ними смешаны в классе. 
ФП предполагает их разделение, которое позволяет посмотреть на каждое отдельно.  
Говоря по миксины - по сути нам нужно только добавить новое поведение.
Мой предыдущий пример уже содержал интерфейс. 
В общем случае в JAva для миксинов используется связка интерфейс+дефолтные методы.
Но встречал я также не мало сообщений, что в Java миксины не настоящие, а приближенные.  
Мне показался не совсем полным для понимания вариант только с JAva, поэтому попробовал реализовать то же самое и через Python.
В каком-то смысле на Java Visitor выносит операцию вовне, а миксины включают их внутрь.

```java
public interface FieldValidatable {

    default void validateField(String field){
        if (field == null || field.isBlank()){
            throw new IllegalArgumentException("Invalid field");
        }
    }
}
```

```java
public class EqualsExpression implements Expression, FieldValidatable {
    private final String field;
    private final String value;

    public EqualsExpression(String field, String value) {
        validateField(field);
        this.field = field;
        this.value = value;
    }
}
```

```python
from __future__ import annotations
from typing import Protocol


class ExpressionVisitor(Protocol):
    def visit_equals(self, expression: Equals) -> str:
        ...

    def visit_greater_than(self, expression: GreaterThan) -> str:
        ...

    def visit_and(self, expression: And) -> str:
        ...


class Expression(Protocol):
    def accept(self, visitor: ExpressionVisitor) -> str:
        ...


class Equals:
    def __init__(self, field: str, value: str) -> None:
        self.field = field
        self.value = value

    def accept(self, visitor: ExpressionVisitor) -> str:
        return visitor.visit_equals(self)


class GreaterThan:
    def __init__(self, field: str, value: int) -> None:
        self.field = field
        self.value = value

    def accept(self, visitor: ExpressionVisitor) -> str:
        return visitor.visit_greater_than(self)


class And:
    def __init__(self, left: Expression, right: Expression) -> None:
        self.left = left
        self.right = right

    def accept(self, visitor: ExpressionVisitor) -> str:
        return visitor.visit_and(self)
```

```python
from expressions_types import Equals, GreaterThan, And


class SqlVisitor:
    def visit_equals(self, expression: Equals) -> str:
        return f"{expression.field} = '{expression.value}'"

    def visit_greater_than(self, expression: GreaterThan) -> str:
        return f"{expression.field} > {expression.value}"

    def visit_and(self, expression: And) -> str:
        left_sql = expression.left.accept(self)
        right_sql = expression.right.accept(self)

        return f"({left_sql} AND {right_sql})"
```

```python
from __future__ import annotations
from typing import Protocol

from mixin_types import FieldValidatableMixin


class ExpressionVisitor(Protocol):
    def visit_equals(self, expression: Equals) -> str:
        ...

    def visit_greater_than(self, expression: GreaterThan) -> str:
        ...

    def visit_and(self, expression: And) -> str:
        ...


class Expression(Protocol):
    def accept(self, visitor: ExpressionVisitor) -> str:
        ...


class Equals(FieldValidatableMixin):
    def __init__(self, field: str, value: str) -> None:
        self.validate_field(field)
        self.field = field
        self.value = value

    def accept(self, visitor: ExpressionVisitor) -> str:
        return visitor.visit_equals(self)


class GreaterThan(FieldValidatableMixin):
    def __init__(self, field: str, value: int) -> None:
        self.validate_field(field)
        self.field = field
        self.value = value

    def accept(self, visitor: ExpressionVisitor) -> str:
        return visitor.visit_greater_than(self)


class And:
    def __init__(self, left: Expression, right: Expression) -> None:
        self.left = left
        self.right = right

    def accept(self, visitor: ExpressionVisitor) -> str:
        return visitor.visit_and(self)
```
