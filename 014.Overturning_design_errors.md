Пример 1  
Пресловутый бренд автомобиля при создании нового vehicle. В ДТО он является вложенным и на нём может быть null. А при заполнении по сути только нужен его id и он обязателен

Было
```java
@Data
public class VehicleDTO {
    @NotEmpty(message = "Обязательное поле")
    private String vehicleName;

    @NotEmpty(message = "Госномер обязателен")
    private String licensePlate;

    private BrandDTO brand;
}
```

```java
@PostMapping("/enterprises/{enterpriseId}/vehicles")
public ResponseEntity<Void> createVehicle(@CurrentManagerId Long managerId,
                                          @PathVariable Long enterpriseId,
                                          @RequestBody @Valid VehicleDTO vehicleDTO,
                                          BindingResult bindingResult) {
    ValidationBindingUtil.Binding(bindingResult);

    Enterprise enterprise = enterpriseService.findEnterpriseForManager(managerId, enterpriseId);

    Vehicle vehicle = vehicleMapper.convertToVehicle(vehicleDTO, enterprise.getTimeZone());

    vehicle.setEnterpriseOwnerOfVehicle(enterprise);

    Long brandId = getBrandIdOrThrow(vehicleDTO);

    vehicleService.save(vehicle, brandId);

    return ResponseEntity.status(HttpStatus.CREATED).build();
}
```

```java
//сама проверка по бренду
private Long getBrandIdOrThrow(VehicleDTO dto) {
    if (dto.getBrand() == null || dto.getBrand().getBrandId() == null) {
        throw new IllegalArgumentException("Не указан бренд автомобиля");
    }
    return dto.getBrand().getBrandId();
}

```
Стало

```java
public record VehicleWriteRequest(
        @NotBlank(message = "Название автомобиля обязательно")

        @Min(value = 0, message = "Стоимость не может быть отрицательной")

        @Min(value = 1900, message = "Год выпуска не может быть меньше 1900")
        int vehicleYearOfRelease,

        @NotNull(message = "Бренд автомобиля обязателен")
        Long brandId,

        String purchaseDateEnterpriseTime
) {  }
```

```java
@PostMapping("/enterprises/{enterpriseId}/vehicles")
public ResponseEntity<Void> createVehicle(@CurrentManagerId Long managerId,
                                          @PathVariable Long enterpriseId,
                                          @RequestBody @Valid VehicleWriteRequest request,
                                          BindingResult bindingResult) {
    ValidationBindingUtil.Binding(bindingResult);
    
    Enterprise enterprise = enterpriseService.findEnterpriseForManager(managerId, enterpriseId);

    vehicle.setEnterpriseOwnerOfVehicle(enterprise);

    vehicleService.save(vehicle, request.brandId());

    return ResponseEntity.status(HttpStatus.CREATED).build();
}
```
